1 · Submit the browser request · part 1

The browser sends a same-origin request and updates the visible evidence. It never receives the API key.

const button = document.querySelector("#run"),
  status = document.querySelector("#status");
fetch("/api/config")
  .then((r) => r.json())
  .then((data) => {
    button.disabled = !data.configured;
    status.textContent = data.configured
      ? "Configuration ready. Your API key stays on the server."
      : "Add your API key to .env.";
  })
  .catch(() => {
    status.textContent = "Could not check configuration.";
  });
button.addEventListener("click", async () => {
  button.disabled = true;
  status.textContent = "Creating and retrieving the environment…";
  try {
    const response = await fetch("/api/environment", { method: "POST" });
    const data = await response.json();
    if (!response.ok) throw new Error(data.error || "Request failed");
    document.querySelector("#envId").textContent = data.createdId;
    document.querySelector("#config").textContent = JSON.stringify(
      data.environment.config,
      null,
      2,
    );
    document.querySelector("#empty").hidden = true;
    document.querySelector("#result").hidden = false;
    status.textContent = "Environment created. Save its ID for reuse.";
  } catch (error) {
    status.textContent = error.message;
  } finally {
    button.disabled = false;
  }
});

2 · Execute the provider workflow · part 1

The Node host validates input, uses configured resources, and returns observable evidence. Read the stream ordering and failure branches.

Source: src/server.ts · starting at line 1

import { serveStudentAsset } from "./student-assets";
import http from "node:http";
import { readFile } from "node:fs/promises";
import { existsSync } from "node:fs";
import path from "node:path";
import Anthropic from "@anthropic-ai/sdk";
import { createRestricted, restrictedConfig } from "./policy";
if (existsSync(".env")) process.loadEnvFile(".env");
const port = Number(process.env.PORT || 3005);
const key = process.env.ANTHROPIC_API_KEY || "";
const configured = Boolean(key && !key.startsWith("your_"));
http.createServer(async (req, res) => {
  res.setHeader("Cache-Control", "no-store");
  try {
    if (await serveStudentAsset(req, res)) return;
    if (req.method === "GET" && req.url === "/api/config") { res.writeHead(200, { "Content-Type": "application/json" }); res.end(JSON.stringify({ configured, policy: restrictedConfig })); return; }
    if (req.method === "POST" && req.url === "/api/environment") {
      const origin = req.headers.origin;
      if (origin && origin !== `http://127.0.0.1:${port}` && origin !== `http://localhost:${port}`) { res.writeHead(403).end(); return; }
      if (!configured) throw new Error("Configure ANTHROPIC_API_KEY in .env.");
      const client = new Anthropic({ apiKey: key });
      const created = await createRestricted(client);
      try {
        const retrieved = await client.beta.environments.retrieve(created.id);
        res.writeHead(200, { "Content-Type": "application/json" }); res.end(JSON.stringify({ createdId: created.id, environment: retrieved }));
      } catch (error) { res.writeHead(502, { "Content-Type": "application/json" }); res.end(JSON.stringify({ error: `Created environment ${created.id}, but retrieval failed: ${error instanceof Error ? error.message : "unknown error"}. Save the ID before retrying.` })); }
      return;
    }

3 · Understand the lesson-specific rules · part 1

Creates a real restricted cloud environment and retrieves its configuration. Every click creates a resource; a failed retrieval is not proof creation failed. This app does not run network probes.

Source: src/policy.ts · starting at line 1

import Anthropic from "@anthropic-ai/sdk";
export const restrictedConfig = {
  type: "cloud",
  networking: {
    type: "limited",
    allowed_hosts: [],
    allow_package_managers: false,
    allow_mcp_servers: false,
  },
} as const;
export async function createRestricted(client: Anthropic) {
  return client.beta.environments.create({
    name: "Lesson C05 restricted environment",
    config: {
      type: "cloud",
      networking: {
        type: "limited",
        allowed_hosts: [],
        allow_package_managers: false,
        allow_mcp_servers: false,
      },
    },
  });
}

What the evidence proves
